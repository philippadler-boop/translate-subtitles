############################################################
# Core runtime dependencies for translate-subtitles project #
############################################################

# Subtitle parsing / writing
srt

# Progress bars (may be transitively installed by other libs but kept explicit)
tqdm

# Optional loading of environment variables (.env) for API keys
python-dotenv

# Translation engines / APIs
deep-translator        # Google + other web-based engines (no key needed for Google)
deepl                  # Official DeepL API SDK (requires DEEPL_AUTH_KEY)
openai                 # GPT-4o / other OpenAI models (requires OPENAI_API_KEY)

# HuggingFace and tokenizer support
transformers           # HF pipelines (used for opus-mt models)
sentencepiece          # Needed for many MT models (e.g. Helsinki-NLP opus-mt)
sacremoses             # Additional tokenizer (optional but commonly needed)

# Backports and packaging helpers to avoid `pkg_resources` usage and support
# migration away from pkg_resources in dependencies.
importlib-resources    # backport of importlib.resources for older Python
importlib-metadata     # backport of importlib.metadata for older Python
packaging              # utilities for version/requirement parsing

# Local ASR: using Hugging Face `transformers` pipeline instead of faster-whisper
# (install `transformers` and matching PyTorch for GPU support)

# Media handling
ffmpeg-python          # ffmpeg invocation helper (extraction)
webrtcvad              # Voice Activity Detection segmentation
